{{- /* The value objectives a milestone serves: its own `value:` list if it has one, otherwise its item's. An id not in data/value_objectives.yaml fails the build. Takes dict "page" and "milestone". */ -}} {{- $values := .page.Params.value | default slice -}} {{- if isset .milestone "value" -}}{{- $values = .milestone.value | default slice -}}{{- end -}} {{- $known := slice -}} {{- range site.Data.value_objectives.objectives -}}{{- $known = $known | append .id -}}{{- end -}} {{- range $values -}} {{- if not (in $known .) -}}{{- errorf "%s: unknown value objective %q" $.page.Path . -}}{{- end -}} {{- end -}} {{- return $values -}}
