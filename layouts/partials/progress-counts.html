{{- /* A page's progress: its milestones (the roadmap) or its hand-counted tasks_* (the plans). A section sums itself and everything beneath it, milestones if any have them, so a roadmap item or area never counts the tasks of the plans under it; otherwise tasks. */ -}} {{- $c := 0 -}}{{- $a := 0 -}}{{- $p := 0 -}} {{- $pages := slice . -}} {{- if .IsSection -}} {{- $under := printf "%s/" .Path -}} {{- range site.Pages -}}{{- if hasPrefix .Path $under -}}{{- $pages = $pages | append . -}}{{- end -}}{{- end -}} {{- end -}} {{- $hasMilestones := false -}} {{- range $pages -}}{{- if .Params.milestones -}}{{- $hasMilestones = true -}}{{- end -}}{{- end -}} {{- range $pages -}} {{- if $hasMilestones -}} {{- range .Params.milestones -}} {{- if eq .status "done" -}}{{- $c = add $c 1 -}} {{- else if eq .status "active" -}}{{- $a = add $a 1 -}} {{- else -}}{{- $p = add $p 1 -}}{{- end -}} {{- end -}} {{- else -}} {{- $c = add $c (.Params.tasks_completed | default 0) -}} {{- $a = add $a (.Params.tasks_in_progress | default 0) -}} {{- $p = add $p (.Params.tasks_planned | default 0) -}} {{- end -}} {{- end -}} {{- return (dict "completed" $c "in_progress" $a "planned" $p) -}}
